cmake_minimum_required(VERSION 3.24)
project(billing_management VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(billing_options INTERFACE)
if(MSVC)
  target_compile_options(billing_options INTERFACE /W4 /WX /permissive-)
else()
  target_compile_options(billing_options INTERFACE -Wall -Wextra -Wpedantic -Werror)
endif()
add_subdirectory(shared/domain)

if(EMSCRIPTEN)
  add_subdirectory(management/frontend)
  add_subdirectory(customer/frontend)
else()
  add_subdirectory(management/backend)
  add_subdirectory(customer/backend)
  include(CTest)
  if(BUILD_TESTING)
    add_subdirectory(tests)
  endif()
endif()
