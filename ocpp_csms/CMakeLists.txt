cmake_minimum_required(VERSION 3.24)
project(ocpp_server_cpp VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
option(OCPP_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
include(cmake/dependencies.cmake)
include(cmake/hardening.cmake)
include(cmake/targets.cmake)
include(cmake/admin-assets.cmake)
enable_testing()
add_test(NAME core COMMAND ocpp_tests ${CMAKE_CURRENT_SOURCE_DIR}/schemas)
find_package(Python3 QUIET COMPONENTS Interpreter)
if(Python3_FOUND)
  add_test(NAME architecture COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/check-architecture.py)
  add_test(NAME deployment_config COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/unit/deploy_tools_test.py)
endif()
install(TARGETS ocpp_server RUNTIME DESTINATION bin COMPONENT Runtime)
install(DIRECTORY schemas DESTINATION share/ocpp COMPONENT Runtime)
