cmake_minimum_required(VERSION 3.24)
project(ev_charger_simulation VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(sim_options INTERFACE)
if(MSVC)
  target_compile_options(sim_options INTERFACE /W4 /WX /permissive- /utf-8 /sdl)
else()
  target_compile_options(sim_options INTERFACE -Wall -Wextra -Wpedantic -Werror -fstack-protector-strong)
  if(NOT EMSCRIPTEN)
    target_link_options(sim_options INTERFACE -Wl,-z,relro,-z,now)
  endif()
endif()
add_library(sim_physics STATIC shared/domain/src/physics.cpp)
target_include_directories(sim_physics PUBLIC shared/domain/include)
target_link_libraries(sim_physics PRIVATE sim_options)
if(EMSCRIPTEN)
  add_executable(sim_estimate shared/wasm/estimate.cpp)
  target_link_libraries(sim_estimate PRIVATE sim_physics sim_options)
  set_target_properties(sim_estimate PROPERTIES OUTPUT_NAME physics SUFFIX .js
    RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/wasm)
  target_link_options(sim_estimate PRIVATE --no-entry -sMODULARIZE=1 -sEXPORT_ES6=1
    -sENVIRONMENT=web -sFILESYSTEM=0 "-sEXPORTED_FUNCTIONS=['_sim_estimate_seconds']")
else()
  include(cmake/dependencies.cmake)
  add_subdirectory(backend)
  enable_testing()
  add_executable(sim_physics_tests tests/unit/physics_tests.cpp)
  target_link_libraries(sim_physics_tests PRIVATE sim_physics sim_options)
  add_test(NAME physics COMMAND sim_physics_tests)
endif()
