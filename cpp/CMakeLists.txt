cmake_minimum_required(VERSION 3.24)
project(ocpp_server_cpp VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(FetchContent)
include(CTest)
option(OCPP_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
set(BUILD_TESTING OFF CACHE BOOL "Dependency tests" FORCE)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
set(JSON_VALIDATOR_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(JSON_VALIDATOR_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(JSON_VALIDATOR_INSTALL OFF CACHE BOOL "" FORCE)
set(BUILD_CTL OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_ORM OFF CACHE BOOL "" FORCE)
set(BUILD_BROTLI OFF CACHE BOOL "" FORCE)
set(BUILD_YAML_CONFIG OFF CACHE BOOL "" FORCE)
FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG 55f93686c01528224f448c19128836e7df245f72)
FetchContent_Declare(schema_validator GIT_REPOSITORY https://github.com/pboettch/json-schema-validator.git GIT_TAG 349cba9f7e3cb423bbc1811bdd9f6770f520b468)
FetchContent_Declare(drogon GIT_REPOSITORY https://github.com/drogonframework/drogon.git GIT_TAG 4c5430757ea5451a7c38fbbef4b4bef7dbb47f2f GIT_SUBMODULES trantor)
FetchContent_Declare(pugixml GIT_REPOSITORY https://github.com/zeux/pugixml.git GIT_TAG c8033ce9d039e7f9d134877c363397b3cfe20816)
FetchContent_MakeAvailable(json schema_validator drogon pugixml)
include(cmake/harden-drogon.cmake)
find_path(MARIADB_INCLUDE_DIR mysql.h PATH_SUFFIXES mariadb mysql REQUIRED)
find_library(MARIADB_LIBRARY NAMES libmariadb mariadb mariadbclient PATH_SUFFIXES mariadb REQUIRED)
add_library(ocpp_core src/protocol.cpp src/security.cpp src/executor.cpp src/mysql.cpp src/service.cpp src/commands.cpp src/tasks.cpp src/ocpp201.cpp src/legacy.cpp src/http_client.cpp src/certificates.cpp src/billing.cpp)
target_include_directories(ocpp_core PUBLIC include PRIVATE ${MARIADB_INCLUDE_DIR})
target_link_libraries(ocpp_core PUBLIC nlohmann_json::nlohmann_json nlohmann_json_schema_validator PRIVATE ${MARIADB_LIBRARY} pugixml::pugixml)
if(WIN32)
  target_link_libraries(ocpp_core PRIVATE winhttp crypt32 bcrypt)
else()
  find_package(CURL 7.85 REQUIRED)
  find_package(OpenSSL 3 REQUIRED)
  target_link_libraries(ocpp_core PRIVATE CURL::libcurl OpenSSL::Crypto)
endif()
add_executable(ocpp_server src/server.cpp)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ui/index.html ui/styles.css ui/app.js)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/ui/index.html" ADMIN_HTML)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/ui/styles.css" ADMIN_CSS)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/ui/app.js" ADMIN_JS)
# MSVC limits each string literal to 16 KiB. Keep UTF-8 assets as adjacent
# newline-sized literals, compiled into the executable without runtime files.
foreach(asset ADMIN_HTML ADMIN_CSS ADMIN_JS)
  string(REPLACE "\n" "\n)OCPPUI\"\nR\"OCPPUI(" ${asset} "${${asset}}")
endforeach()
configure_file(ui/assets.hpp.in generated/admin_assets.hpp @ONLY)
target_include_directories(ocpp_server PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(ocpp_server PRIVATE ocpp_core Drogon::Drogon)
add_executable(ocpp_tests tests/core_tests.cpp)
target_link_libraries(ocpp_tests PRIVATE ocpp_core)
add_executable(ocpp_benchmark tests/benchmark.cpp)
target_link_libraries(ocpp_benchmark PRIVATE ocpp_core)
foreach(target ocpp_core ocpp_server ocpp_tests ocpp_benchmark)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8 /sdl /guard:cf)
    target_link_options(${target} PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -fstack-protector-strong)
    if(UNIX AND NOT APPLE)
      target_link_options(${target} PRIVATE -Wl,-z,relro,-z,now)
    endif()
    if(OCPP_SANITIZERS)
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
endforeach()
enable_testing()
add_test(NAME core COMMAND ocpp_tests ${CMAKE_CURRENT_SOURCE_DIR}/schemas)
install(TARGETS ocpp_server RUNTIME DESTINATION bin)
install(DIRECTORY schemas DESTINATION share/ocpp)
